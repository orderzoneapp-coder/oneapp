async function checkActualReport() {
  const require = (ok, message) => { if (!ok) throw new Error(message); };
  const same = (a,b,message) => require(JSON.stringify(a)===JSON.stringify(b),message);
  const api = await import('/smartinput/purchase-sales-output.js?v=0.1.3');
  const row=(group,customer,code,quantity,cost,extra={})=>({거래처:group,거래처명:customer,코드:code,품명:`검증상품-${code}`,수량:quantity,입고가:cost,구매처:'기타구매',일자:'2026-09-23',...extra});
  const fixtures=[
    row('5온산','5온산','104560112',2,20000,{식자재:3000}),
    row('3우리','마산99번','104550112',2,1000,{도매A:4000}),
    row('1마산','마산75번','104014110',0,1000,{청과:2000}),
    row('4연산','4연산','104560112',3,20000,{식자재:3000}),
    row('1마산','마산99번','104550112',2,1000,{도매A:4000}),
    row('3우리','중앙170','103044110',1,75000,{도매A:51000}),
    row('2중앙','중앙170','103044110',1,75000,{도매A:51000}),
    row('3우리','내부입고가없음','000123',1,0)
  ];
  const before=JSON.stringify(fixtures);
  const output=api.buildPurchaseSalesUploadData(fixtures);
  require(JSON.stringify(fixtures)===before,'Source records were mutated');
  same(output.matrices['확인요청'][0],['이슈','그룹','거래처','품목코드','품명','수량','확인사항'],'Confirmation column contract');
  const checks=output.matrices['확인요청'].slice(1);
  same(checks.map(r=>[r[0],r[3],r[1],r[5]]),[
    ['도매A','104550112','1마산',2],['도매A','104550112','3우리',2],
    ['수량','104014110','1마산',0],
    ['역마진','103044110','2중앙',1],['역마진','103044110','3우리',1],
    ['역마진','104560112','4연산',3],['역마진','104560112','5온산',2],
    ['입고가','000123','3우리',1]
  ],'Expected issue/product/group ordering and preserved quantities');
  require(checks.filter(r=>r[0]==='도매A').every(r=>r[6].includes('1,000') && r[6].includes('4,000')),'Amounts were split at thousands separators');
  const sales=output.matrices['판매입력'].slice(1);
  require(sales.some(r=>r[8]==='103044110' && r[3]==='2중앙' && r[12]===56000),'Existing listing price selection changed');
  require(sales.some(r=>r[8]==='103044110' && r[3]==='3우리' && r[12]===75000),'Existing internal-transfer price changed');
  const mixed=api.buildPurchaseSalesUploadData([row('미등록','복수이슈','MIXED',1,1000,{품명:''})]);
  same(mixed.matrices['확인요청'].slice(1).map(r=>r[0]),['단가그룹','품명'],'Multi-issue discoverability');
  const leadingCodes=api.buildPurchaseSalesUploadData([
    row('3우리','가','001',1,1000,{도매A:4000}),row('1마산','나','1',1,1000,{도매A:4000}),
    row('1마산','나','001',1,1000,{도매A:4000}),row('3우리','가','1',1,1000,{도매A:4000})
  ]);
  same(leadingCodes.matrices['확인요청'].slice(1).map(r=>r[3]),['001','001','1','1'],'Exact SKU identity');
  const bulkRows=Array.from({length:600},(_,i)=>({...fixtures[i%fixtures.length],코드:`VERIFY-${String(i).padStart(4,'0')}`}));
  const metrics=[];
  const large=await api.runStage5Compute({feature:'purchase-sales-report',phase:'PURCHASE_SALES_BUILD',payload:{rows:bulkRows},rowCount:bulkRows.length,direct:()=>api.buildPurchaseSalesUploadData(bulkRows),onMetric:m=>metrics.push(m)});
  require(metrics.at(-1)?.path==='worker','Actual production Worker required; fallback is not accepted');
  same(large,api.buildPurchaseSalesUploadData(bulkRows),'Production Worker/direct parity');
  require(large.matrices['판매입력'].length===601 && large.matrices['확인요청'].length===601,'Transaction or issue rows lost');
  const smallMetrics=[];
  await api.runStage5Compute({feature:'purchase-sales-report',phase:'PURCHASE_SALES_BUILD',payload:{rows:fixtures},rowCount:fixtures.length,direct:()=>api.buildPurchaseSalesUploadData(fixtures),onMetric:m=>smallMetrics.push(m)});
  require(smallMetrics.at(-1)?.path==='direct','Small report direct path');
  if (!window.XLSX) await new Promise((resolve,reject)=>{
    const script=document.createElement('script');script.src='/customer-master/vendor/xlsx.full.min.js';script.onload=resolve;script.onerror=()=>reject(new Error('Production XLSX script failed'));document.head.appendChild(script);
  });
  const book=XLSX.utils.book_new();
  output.sheetNames.forEach(name=>{const sheet=XLSX.utils.aoa_to_sheet(output.matrices[name]);sheet['!cols']=output.widths[name].map(wch=>({wch}));XLSX.utils.book_append_sheet(book,sheet,name);});
  const bytes=XLSX.write(book,{type:'array',bookType:'xlsx'});
  const reopened=XLSX.read(new Uint8Array(bytes),{type:'array'});
  same(reopened.SheetNames,output.sheetNames,'Seven worksheet names and order');
  for(const name of output.sheetNames) same(XLSX.utils.sheet_to_json(reopened.Sheets[name],{header:1,raw:true,defval:'',blankrows:false}),output.matrices[name],`XLSX round trip: ${name}`);
  const estimate=await import('/smartinput/estimate-output.js?v=0.2.10');
  const promo=estimate.buildEstimateF8Data([{itemCode:'VERIFY-PROMO',itemName:'검증행',outPrice:'',promoPrice:15800}]);
  require(promo.shopData[1][14]==='1','Previous promotion-only shop sale fix regressed');
  return {issueOrder:[...new Set(checks.map(r=>r[0]))],confirmationRows:checks.length,example:checks.map(r=>[r[0],r[3],r[1],r[5]]),bulkRows:600,workerPath:metrics.at(-1).path,smallPath:smallMetrics.at(-1).path,xlsxSheetsVerified:reopened.SheetNames,sourcePreserved:true,existingPricesPreserved:true,location:location.href};
}
